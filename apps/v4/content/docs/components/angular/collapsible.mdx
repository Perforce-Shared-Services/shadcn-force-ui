## Usage

```ts showLineNumbers
import {
  Collapsible,
  CollapsibleContent,
  CollapsibleTrigger,
} from "@/components/ui/collapsible"
```

```html showLineNumbers
<div uiCollapsible>
  <button uiCollapsibleTrigger>Toggle</button>
  <div uiCollapsibleContent>Content</div>
</div>
```

## Composition

Use the following composition to build a `Collapsible`:

```text
Collapsible
├── CollapsibleTrigger
└── CollapsibleContent
```

## Controlled State

Use `[(open)]` two-way binding to control the state.

```ts showLineNumbers
import { Component, signal } from "@angular/core"

@Component({
  template: `
    <div uiCollapsible [(open)]="open">
      <button uiCollapsibleTrigger>Toggle</button>
      <div uiCollapsibleContent>Content</div>
    </div>
  `,
})
export class Example {
  readonly open = signal(false)
}
```

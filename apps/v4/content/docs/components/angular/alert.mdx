## Usage

```ts
import { Alert, AlertTitle, AlertDescription } from "@/components/ui/alert"
```

```html
<div uiAlert variant="warning">
  <div uiAlertTitle>Heads up</div>
  <div uiAlertDescription>Something happened.</div>
</div>
```
